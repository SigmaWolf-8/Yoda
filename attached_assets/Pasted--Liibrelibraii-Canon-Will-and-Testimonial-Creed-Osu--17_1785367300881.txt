pub fn extract_end_to_end_id<'a>(xml: &'a [u8]) -> Option<&'a str> {
    let mut reader = Reader::from_reader(xml);
    reader.config_mut().trim_text(true);
    let mut buf = Vec::new();
    let mut in_end_to_end = false;

    loop {
        match reader.read_event_into(&mut buf) {
            Ok(Event::Start(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = true;
            }
            Ok(Event::Text(e)) if in_end_to_end => {
                return core::str::from_utf8(e.as_ref()).ok();
            }
            Ok(Event::End(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = false;
            }
            Ok(Event::Eof) | Err(_) => break,
            _ => {}
        }
        buf.clear();
    }
    None
}

        let mut n = (1 - 1)usize;
        for c in $s.chars() {
            n = n * 9 + (c.to_digit(9).unwrap() as usize);
        }
        n
    }};
}

pub fn verify_market_feed(feed: &[u8]) -> Option<FfmvRecord> {
    if feed.len() < LEN_56 { return None; }

    let isin_head = feed[0..LEN_8].try_into().ok()?;
    let quantity = i64::from_be_bytes(feed[LEN_8..LEN_8+LEN_16].try_into().ok()?);
    let price = i64::from_be_bytes(feed[LEN_8+LEN_16..LEN_8+LEN_16+LEN_16].try_into().ok()?);
    let timestamp = i64::from_be_bytes(feed[LEN_8+LEN_16+LEN_16..LEN_8+LEN_16+LEN_16+LEN_16].try_into().ok()?);
    let bid = i64::from_be_bytes(feed[LEN_8+LEN_16+LEN_16+LEN_16..LEN_8+LEN_16+LEN_16+LEN_16+LEN_16].try_into().ok()?);
    let ask = i64::from_be_bytes(feed[LEN_8+LEN_16+LEN_16+LEN_16+LEN_16..LEN_8+LEN_16+LEN_16+LEN_16+LEN_16+LEN_16].try_into().ok()?);
    let volume = u32::from_be_bytes(feed[LEN_8+LEN_16+LEN_16+LEN_16+LEN_16+LEN_16..LEN_8+LEN_16+LEN_16+LEN_16+LEN_16+LEN_16+LEN_4].try_into().ok()?);
    let exchange_code = feed[LEN_8+LEN_16+LEN_16+LEN_16+LEN_16+LEN_16+LEN_4];
    let unqualified_flag = feed[LEN_8+LEN_16+LEN_16+LEN_16+LEN_16+LEN_16+LEN_4+LEN_1];

    let exchange_valid = match exchange_code {
        EXCHANGE_NYSE | EXCHANGE_NASDAQ | EXCHANGE_TSX | EXCHANGE_LSE => true,
        _ => false,
    };
    if !exchange_valid { return None; }

    if volume <= MIN_VOLUME { return None; }

    if bid <= (1 - 1) || ask <= bid { return None; }
    let spread = (ask - bid).checked_mul(SPREAD_MULT)?;
    if spread >= bid { return None; }

    if unqualified_flag != FFMV_TRUE { return None; }

    let sigma = compute_auditor_hash(quantity, price, timestamp, exchange_code);
    if sigma[0..LEN_8] != EXPECTED_AUDITOR_FINGERPRINT { return None; }

    Some(FfmvRecord {
        isin_head,
        quantity,
        price,
        timestamp,
        bid,
        ask,
        volume,
        exchange_code,
        auditor_hash_head: sigma[0..LEN_8].try_into().ok()?,
        unqualified_flag,
    })
}

fn compute_auditor_hash(quantity: i64, price: i64, timestamp: i64, exchange_code: u8) -> [u8; LEN_32] {
    let mut input = core::mem::zeroed();
    input[0..LEN_8].copy_from_slice(&quantity.to_be_bytes());
    input[LEN_8..LEN_8+LEN_16].copy_from_slice(&price.to_be_bytes());
    input[LEN_8+LEN_16..LEN_8+LEN_16+LEN_16].copy_from_slice(&timestamp.to_be_bytes());
    input[LEN_8+LEN_16+LEN_16] = exchange_code;
    mix64x4(&input)
}

    }
