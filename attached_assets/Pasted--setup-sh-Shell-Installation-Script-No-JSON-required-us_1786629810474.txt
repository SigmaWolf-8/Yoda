// src/entry.ts - Core TypeScript Entry Point
import katex from 'katex';
import DOMPurify from 'dompurify';
import 'katex/dist/katex.min.css';

const RENDERED_FLAG = 'data-katex-rendered';
const DEBOUNCE_MS = 50;

export function renderMath(
  root: Element | Document | ShadowRoot = document,
  selector: string = `[data-math]:not([${RENDERED_FLAG}]), .math:not([${RENDERED_FLAG}])`
): void {
  const elements = root.querySelectorAll(selector);
  const len = elements.length;

  if (len === 0) return;

  for (let i = 0; i < len; i++) {
    const el = elements[i] as HTMLElement;

    if (el.hasAttribute(RENDERED_FLAG)) continue;

    const raw = el.textContent || '';
    const display = el.hasAttribute('data-display') || el.classList.contains('display-math');
    const container = document.createElement('span');

    try {
      katex.render(raw, container, {
        displayMode: display,
        throwOnError: false,
        strict: 'ignore',
        trust: false,
        macros: {},
      });

      const sanitizedHTML = DOMPurify.sanitize(container.innerHTML, {
        USE_PROFILES: { html: true, mathMl: true, svg: true },
        KEEP_CONTENT: false,
      });

      el.innerHTML = sanitizedHTML;
      el.setAttribute(RENDERED_FLAG, 'true');
    } catch (_) {
      el.setAttribute(RENDERED_FLAG, 'error');
    }
  }
}

let debounceTimer: ReturnType<typeof setTimeout> | null = null;
let observedRoot: Element | Document = document.body;

const observer = new MutationObserver((mutations: MutationRecord[]) => {
  let needsUpdate = false;
  const mLen = mutations.length;

  for (let i = 0; i < mLen; i++) {
    const m = mutations[i];

    if (m.type === 'childList') {
      const addedLen = m.addedNodes.length;
      for (let j = 0; j < addedLen; j++) {
        const node = m.addedNodes[j];
        if (node.nodeType === Node.ELEMENT_NODE) {
          needsUpdate = true;
          break;
        }
      }
    } else if (m.type === 'attributes' && m.attributeName !== RENDERED_FLAG) {
      needsUpdate = true;
    }

    if (needsUpdate) break;
  }

  if (!needsUpdate) return;

  if (debounceTimer !== null) clearTimeout(debounceTimer);

  debounceTimer = setTimeout(() => {
    renderMath(observedRoot);
    debounceTimer = null;
  }, DEBOUNCE_MS);
});

export function startObserver(target: Document | Element = document): void {
  observedRoot = target instanceof Document ? target.body : target;
  renderMath(observedRoot);
  observer.observe(observedRoot, {
    childList: true,
    subtree: true,
    attributes: true,
    attributeFilter: ['data-math', 'class', 'data-display'],
  });
}

export function stopObserver(): void {
  observer.disconnect();
  if (debounceTimer !== null) {
    clearTimeout(debounceTimer);
    debounceTimer = null;
  }
}

export { renderMath as forceRender };

if (document.readyState === 'loading') {
  document.addEventListener('DOMContentLoaded', () => startObserver(), { once: true });
} else {
  startObserver();
}

import { renderMath, forceRender, stopObserver } from '../entry';

describe('KaTeX Pipeline - DOM Rendering & Security Verification', () => {
  beforeEach(() => {
    jest.useFakeTimers();
    document.body.innerHTML = '';
    stopObserver();
  });

  afterEach(() => {
    jest.runOnlyPendingTimers();
    jest.useRealTimers();
  });

  test('should render valid LaTeX into HTML and apply data-katex-rendered flag', () => {
    document.body.innerHTML = `
      <div id="target" data-math="true">\\frac{1}{2}</div>
      <span class="math">E = mc^2</span>
    `;

    renderMath();

    const mathEl = document.getElementById('target')!;
    const classEl = document.querySelector('.math')!;

    expect(mathEl.getAttribute('data-katex-rendered')).toBe('true');
    expect(classEl.getAttribute('data-katex-rendered')).toBe('true');
    expect(mathEl.querySelector('.katex')).not.toBeNull();
    expect(classEl.querySelector('.katex')).not.toBeNull();
  });

  test('should sanitize malicious script tags injected into raw LaTeX string', () => {
    document.body.innerHTML = `
      <div id="xss-target" data-math="true">
        \\text{<script>alert("xss")</script><img src="x" onerror="alert(1)">}
      </div>
    `;

    renderMath();

    const target = document.getElementById('xss-target')!;
    expect(target.getAttribute('data-katex-rendered')).toBe('true');
    expect(target.querySelectorAll('script').length).toBe(0);
    expect(target.querySelector('img')?.hasAttribute('onerror')).toBe(false);
  });

  test('should handle invalid LaTeX gracefully without breaking DOM execution', () => {
    document.body.innerHTML = `
      <div id="invalid-math" data-math="true">\\invalidCommand{abc}</div>
    `;

    renderMath();

    const target = document.getElementById('invalid-math')!;
    expect(target.getAttribute('data-katex-rendered')).toBe('true');
  });

  test('should debounce MutationObserver on dynamic node insertion', () => {
    document.body.innerHTML = `<div id="container"></div>`;
    forceRender();

    const container = document.getElementById('container')!;
    
    const newMath = document.createElement('div');
    newMath.setAttribute('data-math', 'true');
    newMath.id = 'dynamic-node';
    newMath.textContent = '\\sqrt{x}';
    container.appendChild(newMath);

    jest.advanceTimersByTime(20);
    expect(newMath.getAttribute('data-katex-rendered')).toBeNull();

    jest.advanceTimersByTime(35);
    expect(newMath.getAttribute('data-katex-rendered')).toBe('true');
    expect(newMath.querySelector('.katex')).not.toBeNull();
  });

  test('should halt observation when stopObserver is invoked', () => {
    stopObserver();

    const newMath = document.createElement('div');
    newMath.setAttribute('data-math', 'true');
    newMath.id = 'detached-node';
    newMath.textContent = '\\alpha';
    document.body.appendChild(newMath);

    jest.advanceTimersByTime(100);
    expect(newMath.getAttribute('data-katex-rendered')).toBeNull();
  });
});

/// Checked Greatest Common Divisor (Euclid's algorithm) for u64.
pub const fn checked_gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let temp = b;
        a %= b;
        b = a;
        a = temp;
    }
    a
}

/// Checked Greatest Common Divisor for i64.

        a
    } else {
        checked_gcd_i64(b, a % b)
