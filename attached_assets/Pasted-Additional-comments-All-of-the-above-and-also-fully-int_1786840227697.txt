#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct BijectiveBase9 {
    digits: [u8; MAX_BIJECTIVE_DIGITS],
    len: usize,
    raw_value: u64,
}

impl BijectiveBase9 {
    pub const Z_ZERO: Self = Self {
        digits: [0; MAX_BIJECTIVE_DIGITS],
        len: 0,
        raw_value: 0,
    };

    /// Encodes any `u64` into Bijective Base-9.
    /// Contract: never overflows the digit buffer for any input in `0..=u64::MAX`.
    /// The maximum number of bijective-base-9 digits of a 64-bit value is 20
    /// (because \(9^{20} > 2^{64}-1 > 9^{19}\)).
    #[cfg_attr(feature = "prusti", requires(true))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits() && b.raw() == value
    }))]
    #[cfg_attr(feature = "prusti", ensures(result.is_err() ==> matches!(result, Err(EngineError::IntegerOverflow))))]
    pub fn from_u64(mut value: u64) -> Result<Self, EngineError> {
        if value == 0 {
            return Ok(Self::Z_ZERO);
        }

        let orig = value;
        let mut digits = [0u8; MAX_BIJECTIVE_DIGITS];
        let mut idx = 0usize;

        while value > 0 {
            if idx >= MAX_BIJECTIVE_DIGITS {
                // Unreachable for any u64: 9^20 exceeds 2^64-1
                return Err(EngineError::IntegerOverflow);
            }
            // (value-1) % 9 is always safe: value >= 1
            let remainder = (value - 1) % BASE_MODULUS;
            let digit = (remainder + 1) as u8; // 1..=9 by construction
            digits[idx] = digit;
            // Exact integer division; no overflow possible because digit <= 9 and value decreases
            value = (value - (digit as u64)) / BASE_MODULUS;
            idx += 1;
        }

        Ok(Self {
            digits,
            len: idx,
            raw_value: orig,
        })
    }

    #[cfg_attr(feature = "prusti", requires(digits.len() <= MAX_BIJECTIVE_DIGITS))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits()
    }))]
    pub fn from_bijective_digits(digits: &[u8]) -> Result<Self, EngineError> {
        if digits.is_empty() {
            return Ok(Self::Z_ZERO);
        }

        let mut acc: u64 = 0;
        let mut multiplier: u64 = 1;
        let mut stored_digits = [0u8; MAX_BIJECTIVE_DIGITS];

        for (i, &d) in digits.iter().enumerate() {
            if i >= MAX_BIJECTIVE_DIGITS {
                return Err(EngineError::IntegerOverflow);
            }
            if !(1..=9).contains(&d) {
                return Err(EngineError::InvalidDigit(d));
            }
            stored_digits[i] = d;

            let term = match (d as u64).checked_mul(multiplier) {
                Some(t) => t,
                None => return Err(EngineError::IntegerOverflow),
            };
            acc = match acc.checked_add(term) {
                Some(a) => a,
                None => return Err(EngineError::IntegerOverflow),
            };

            if i + 1 < digits.len() {
                multiplier = match multiplier.checked_mul(BASE_MODULUS) {
                    Some(m) => m,
                    None => return Err(EngineError::IntegerOverflow),
                };
            }
        }

        Ok(Self {
            digits: stored_digits,
            len: digits.len(),
            raw_value: acc,
        })
    }

    pub const fn raw(&self) -> u64 {
        self.raw_value
    }

    pub fn digits(&self) -> &[u8] {
        &self.digits[..self.len]
    }

    pub fn verify_digits(&self) -> bool {
        if self.raw_value == 0 {
            return self.len == 0;
        }
        for i in 0..self.len {
            if self.digits[i] < 1 || self.digits[i] > 9 {
                return false;
            }
        }
        true
    }

    pub fn gcd_9_invariant(&self) -> u64 {
        checked_gcd(self.raw_value, BASE_MODULUS)
    }

    pub fn digital_root_invariant(&self) -> u64 {
        if self.raw_value == 0 {
            0
        } else {
            ((self.raw_value - 1) % BASE_MODULUS) + 1
        }
    }
}

/// Checked GCD (Euclid). Never overflows.
pub const fn checked_gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let temp = b;
        a %= b;
        b = a;
        a = temp;
    }
    a
}

pub fn checked_gcd_i64(a: i64, b: i64) -> i64 {
    if b == 0 {
        a
    } else {
        checked_gcd_i64(b, a % b)
    }
}

    pub den: i64,

impl RationalMatrix2x2 {
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let s = result.unwrap();
        s.den != 0
    }))]
    pub fn simplify(&self) -> Result<Self, EngineError> {
        if self.den == 0 {
            return Err(EngineError::DivisionByZero);
        }
        let mut g = self.den.abs();
        for row in &self.num {
            for &val in row {
                g = checked_gcd_i64(g, val.abs());
            }
        }
        if g == 0 {
            return Err(EngineError::DivisionByZero);
        }
        Ok(Self {
            num: [
                [self.num[0][0] / g, self.num[0][1] / g],
                [self.num[1][0] / g, self.num[1][1] / g],
            ],
            den: self.den / g,
        })
    }

    pub fn checked_mul(&self, rhs: &Self) -> Result<Self, EngineError> {
        let common_den = self.den.checked_mul(rhs.den).ok_or(EngineError::IntegerOverflow)?;
        let mut res_num = [[0i64; 2]; 2];
        for i in 0..2 {
            for j in 0..2 {
                let mut sum: i64 = 0;
                for k in 0..2 {
                    let prod = self.num[i][k]
                        .checked_mul(rhs.num[k][j])
                        .ok_or(EngineError::IntegerOverflow)?;
                    sum = sum.checked_add(prod).ok_or(EngineError::IntegerOverflow)?;
                }
                res_num[i][j] = sum;
            }
        }
        Self { num: res_num, den: common_den }.simplify()
    }

    pub fn is_idempotent(&self) -> Result<bool, EngineError> {
        let simplified = self.simplify()?;
        let squared = simplified.checked_mul(&simplified)?;
        Ok(squared == simplified)
    }
}

        Self { hash_bytes }
    }

    /// Always returns exactly 243 (Milesian sum). No overflow path.
    pub fn compute_fx(&self) -> u64 {
        let mut word_sum: u64 = 0;
        for chunk in self.hash_bytes.chunks_exact(8) {
            let val = u64::from_be_bytes([
                chunk[0], chunk[1], chunk[2], chunk[3],
                chunk[4], chunk[5], chunk[6], chunk[7],
            ]);
            word_sum = word_sum.wrapping_add(val); // wrapping is intentional & bounded
        }
        let rem = word_sum % MILESIAN_SALVI_SUM;
        MILESIAN_SALVI_SUM - rem + rem // ≡ 243
    }

    pub fn verify_state_seal(&self, on_chain_state: u64) -> Result<bool, EngineError> {
        let fx = self.compute_fx();
        if fx != on_chain_state {
            return Err(EngineError::HashVerificationFailed);
        }
        let delta = fx.checked_sub(on_chain_state).ok_or(EngineError::IntegerOverflow)?;
        Ok(delta == 0)
    }
}

            let digit = (remainder + 1) as u8; // 1..=9

            idx += 1;
        }

        Ok(Self {
            digits,
            len: idx,
            raw_value: orig,
        })
    }

pub const fn checked_gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let temp = b;
        a %= b;
        b = a;
        a = temp;
    }
    a
}

        let mut g = self.den.abs();
        for row in &self.num {

                [self.num[1][0] / g, self.num[1][1] / g],
            ],
            den: self.den / g,
        })
    }

        let mut res_num = [[0i64; 2]; 2];

                    let prod = self.num[i][k].checked_mul(rhs.num[k][j]).ok_or(EngineError::IntegerOverflow)?;

pub const MAX_BIJECTIVE_DIGITS: usize = 20;
pub const BASE_MODULUS: u64 = 9;
pub const MILESIAN_SALVI_SUM: u64 = 243;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum EngineError {
    InvalidDigit(u8),
    IntegerOverflow,
    DivisionByZero,
    NonIdempotentState,
    NonZeroBalanceViolation,
    Base3UniquenessViolation,
    HashVerificationFailed,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct BijectiveBase9 {
    digits: [u8; MAX_BIJECTIVE_DIGITS],
    len: usize,
    raw_value: u64,
}

impl BijectiveBase9 {
    pub const Z_ZERO: Self = Self {
        digits: [0; MAX_BIJECTIVE_DIGITS],
        len: 0,
        raw_value: 0,
    };

    /// Encodes any `u64` into Bijective Base-9.
    /// Contract: never overflows the digit buffer for any input in `0..=u64::MAX`.
    /// The maximum number of bijective-base-9 digits of a 64-bit value is 20
    /// (because \(9^{20} > 2^{64}-1 > 9^{19}\)).
    #[cfg_attr(feature = "prusti", requires(true))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits() && b.raw() == value
    }))]
    #[cfg_attr(feature = "prusti", ensures(result.is_err() ==> matches!(result, Err(EngineError::IntegerOverflow))))]
    pub fn from_u64(mut value: u64) -> Result<Self, EngineError> {
        if value == 0 {
            return Ok(Self::Z_ZERO);
        }

        let orig = value;
        let mut digits = [0u8; MAX_BIJECTIVE_DIGITS];
        let mut idx = 0usize;

        while value > 0 {
            if idx >= MAX_BIJECTIVE_DIGITS {
                // Unreachable for any u64: 9^20 exceeds 2^64-1
                return Err(EngineError::IntegerOverflow);
            }
            // (value-1) % 9 is always safe: value >= 1
            let remainder = (value - 1) % BASE_MODULUS;
            let digit = (remainder + 1) as u8; // 1..=9 by construction
            digits[idx] = digit;
            // Exact integer division; no overflow possible because digit <= 9 and value decreases
            value = (value - (digit as u64)) / BASE_MODULUS;
            idx += 1;
        }

        Ok(Self {
            digits,
            len: idx,
            raw_value: orig,
        })
    }

    #[cfg_attr(feature = "prusti", requires(digits.len() <= MAX_BIJECTIVE_DIGITS))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits()
    }))]
    pub fn from_bijective_digits(digits: &[u8]) -> Result<Self, EngineError> {
        if digits.is_empty() {
            return Ok(Self::Z_ZERO);
        }

        let mut acc: u64 = 0;
        let mut multiplier: u64 = 1;
        let mut stored_digits = [0u8; MAX_BIJECTIVE_DIGITS];

        for (i, &d) in digits.iter().enumerate() {
            if i >= MAX_BIJECTIVE_DIGITS {
                return Err(EngineError::IntegerOverflow);
            }
            if !(1..=9).contains(&d) {
                return Err(EngineError::InvalidDigit(d));
            }
            stored_digits[i] = d;

            let term = match (d as u64).checked_mul(multiplier) {
                Some(t) => t,
                None => return Err(EngineError::IntegerOverflow),
            };
            acc = match acc.checked_add(term) {
                Some(a) => a,
                None => return Err(EngineError::IntegerOverflow),
            };

            if i + 1 < digits.len() {
                multiplier = match multiplier.checked_mul(BASE_MODULUS) {
                    Some(m) => m,
                    None => return Err(EngineError::IntegerOverflow),
                };
            }
        }

        Ok(Self {
            digits: stored_digits,
            len: digits.len(),
            raw_value: acc,
        })
    }

    pub const fn raw(&self) -> u64 {
        self.raw_value
    }

    pub fn digits(&self) -> &[u8] {
        &self.digits[..self.len]
    }

    pub fn verify_digits(&self) -> bool {
        if self.raw_value == 0 {
            return self.len == 0;
        }
        for i in 0..self.len {
            if self.digits[i] < 1 || self.digits[i] > 9 {
                return false;
            }
        }
        true
    }

    pub fn gcd_9_invariant(&self) -> u64 {
        checked_gcd(self.raw_value, BASE_MODULUS)
    }

    pub fn digital_root_invariant(&self) -> u64 {
        if self.raw_value == 0 {
            0
        } else {
            ((self.raw_value - 1) % BASE_MODULUS) + 1
        }
    }
}

/// Checked GCD (Euclid). Never overflows.
pub const fn checked_gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let temp = b;
        a %= b;
        b = a;
        a = temp;
    }
    a
}

pub fn checked_gcd_i64(a: i64, b: i64) -> i64 {
    if b == 0 {
        a
    } else {
        checked_gcd_i64(b, a % b)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct RationalMatrix2x2 {
    pub num: [[i64; 2]; 2],
    pub den: i64,
}

impl RationalMatrix2x2 {
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let s = result.unwrap();
        s.den != 0
    }))]
    pub fn simplify(&self) -> Result<Self, EngineError> {
        if self.den == 0 {
            return Err(EngineError::DivisionByZero);
        }
        let mut g = self.den.abs();
        for row in &self.num {
            for &val in row {
                g = checked_gcd_i64(g, val.abs());
            }
        }
        if g == 0 {
            return Err(EngineError::DivisionByZero);
        }
        Ok(Self {
            num: [
                [self.num[0][0] / g, self.num[0][1] / g],
                [self.num[1][0] / g, self.num[1][1] / g],
            ],
            den: self.den / g,
        })
    }

    pub fn checked_mul(&self, rhs: &Self) -> Result<Self, EngineError> {
        let common_den = self.den.checked_mul(rhs.den).ok_or(EngineError::IntegerOverflow)?;
        let mut res_num = [[0i64; 2]; 2];
        for i in 0..2 {
            for j in 0..2 {
                let mut sum: i64 = 0;
                for k in 0..2 {
                    let prod = self.num[i][k]
                        .checked_mul(rhs.num[k][j])
                        .ok_or(EngineError::IntegerOverflow)?;
                    sum = sum.checked_add(prod).ok_or(EngineError::IntegerOverflow)?;
                }
                res_num[i][j] = sum;
            }
        }
        Self { num: res_num, den: common_den }.simplify()
    }

    pub fn is_idempotent(&self) -> Result<bool, EngineError> {
        let simplified = self.simplify()?;
        let squared = simplified.checked_mul(&simplified)?;
        Ok(squared == simplified)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct TransactionHashSeal {
    pub hash_bytes: [u8; 32],
}

impl TransactionHashSeal {
    pub fn new(hash_bytes: [u8; 32]) -> Self {
        Self { hash_bytes }
    }

    /// Always returns exactly 243 (Milesian sum). No overflow path.
    pub fn compute_fx(&self) -> u64 {
        let mut word_sum: u64 = 0;
        for chunk in self.hash_bytes.chunks_exact(8) {
            let val = u64::from_be_bytes([
                chunk[0], chunk[1], chunk[2], chunk[3],
                chunk[4], chunk[5], chunk[6], chunk[7],
            ]);
            word_sum = word_sum.wrapping_add(val); // wrapping is intentional & bounded
        }
        let rem = word_sum % MILESIAN_SALVI_SUM;
        MILESIAN_SALVI_SUM - rem + rem // ≡ 243
    }

    pub fn verify_state_seal(&self, on_chain_state: u64) -> Result<bool, EngineError> {
        let fx = self.compute_fx();
        if fx != on_chain_state {
            return Err(EngineError::HashVerificationFailed);
        }
        let delta = fx.checked_sub(on_chain_state).ok_or(EngineError::IntegerOverflow)?;
        Ok(delta == 0)
    }
}

pub fn verify_zero_state_balance(
    cost_cad: u64,
    market_cad: u64,
    clifford_correction: u64,
) -> bool {
    if cost_cad < market_cad {
        return false;
    }
    cost_cad
        .checked_sub(market_cad)
        .and_then(|d| d.checked_sub(clifford_correction))
        .map_or(false, |rem| rem == 1)
}

    /// Encodes any `u64` into Bijective Base‑9.  
    /// Contract: never overflows the digit buffer for any input in `0..=u64::MAX`.  
    /// The maximum number of bijective‑base‑9 digits of a 64‑bit value is 20  
    /// (because \(9^{20} > 2^{64}-1 > 9^{19}\)).
    #[cfg_attr(feature = "prusti", requires(true))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits() && b.raw() == value
    }))]
    #[cfg_attr(feature = "prusti", ensures(result.is_err() ==> matches!(result, Err(EngineError::IntegerOverflow))))]
    pub fn from_u64(mut value: u64) -> Result<Self, EngineError> {
        if value == 0 {
            return Ok(Self::Z_ZERO);
        }

        let orig = value;
        let mut digits = [0u8; MAX_BIJECTIVE_DIGITS];
        let mut idx = 0usize;

        while value > 0 {
            if idx >= MAX_BIJECTIVE_DIGITS {
                // Unreachable for any u64: 9^20 exceeds 2^64-1
                return Err(EngineError::IntegerOverflow);
            }
            let remainder = (value - 1) % BASE_MODULUS;
            let digit = (remainder + 1) as u8; // 1..=9 by construction
            digits[idx] = digit;
            value = (value - (digit as u64)) / BASE_MODULUS;
            idx += 1;
        }

        Ok(Self {
            digits,
            len: idx,
            raw_value: orig,
        })
    }

    #[cfg_attr(feature = "prusti", requires(digits.len() <= MAX_BIJECTIVE_DIGITS))]
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let b = result.unwrap();
        b.verify_digits()
    }))]
    pub fn from_bijective_digits(digits: &[u8]) -> Result<Self, EngineError> {
        if digits.is_empty() {
            return Ok(Self::Z_ZERO);
        }

        let mut acc: u64 = 0;
        let mut multiplier: u64 = 1;
        let mut stored_digits = [0u8; MAX_BIJECTIVE_DIGITS];

        for (i, &d) in digits.iter().enumerate() {
            if i >= MAX_BIJECTIVE_DIGITS {
                return Err(EngineError::IntegerOverflow);
            }
            if !(1..=9).contains(&d) {
                return Err(EngineError::InvalidDigit(d));
            }
            stored_digits[i] = d;

            let term = (d as u64).checked_mul(multiplier).ok_or(EngineError::IntegerOverflow)?;
            acc = acc.checked_add(term).ok_or(EngineError::IntegerOverflow)?;

            if i + 1 < digits.len() {
                multiplier = multiplier.checked_mul(BASE_MODULUS).ok_or(EngineError::IntegerOverflow)?;
            }
        }

        Ok(Self {
            digits: stored_digits,
            len: digits.len(),
            raw_value: acc,
        })
    }

    pub const fn raw(&self) -> u64 { self.raw_value }

    pub fn digits(&self) -> &[u8] { &self.digits[..self.len] }

    pub fn verify_digits(&self) -> bool {
        if self.raw_value == 0 {
            return self.len == 0;
        }
        for i in 0..self.len {
            if self.digits[i] < 1 || self.digits[i] > 9 {
                return false;
            }
        }
        true
    }

    pub fn gcd_9_invariant(&self) -> u64 {
        checked_gcd(self.raw_value, BASE_MODULUS)
    }

    pub fn digital_root_invariant(&self) -> u64 {
        if self.raw_value == 0 {
            0
        } else {
            ((self.raw_value - 1) % BASE_MODULUS) + 1
        }
    }
}

/// Checked Euclidean GCD. Never overflows.
pub const fn checked_gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        let temp = b;
        a %= b;
        b = a;
        a = temp;
    }
    a
}

pub fn checked_gcd_i64(a: i64, b: i64) -> i64 {
    if b == 0 { a } else { checked_gcd_i64(b, a % b) }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct RationalMatrix2x2 {
    pub num: [[i64; 2]; 2],
    pub den: i64,
}

impl RationalMatrix2x2 {
    #[cfg_attr(feature = "prusti", ensures(result.is_ok() ==> {
        let s = result.unwrap();
        s.den != 0
    }))]
    pub fn simplify(&self) -> Result<Self, EngineError> {
        if self.den == 0 {
            return Err(EngineError::DivisionByZero);
        }
        let mut g = self.den.abs();
        for row in &self.num {
            for &val in row {
                g = checked_gcd_i64(g, val.abs());
            }
        }
        if g == 0 {
            return Err(EngineError::DivisionByZero);
        }
        Ok(Self {
            num: [
                [self.num[0][0] / g, self.num[0][1] / g],
                [self.num[1][0] / g, self.num[1][1] / g],
            ],
            den: self.den / g,
        })
    }

    pub fn checked_mul(&self, rhs: &Self) -> Result<Self, EngineError> {
        let common_den = self.den.checked_mul(rhs.den).ok_or(EngineError::IntegerOverflow)?;
        let mut res_num = [[0i64; 2]; 2];
        for i in 0..2 {
            for j in 0..2 {
                let mut sum: i64 = 0;
                for k in 0..2 {
                    let prod = self.num[i][k]
                        .checked_mul(rhs.num[k][j])
                        .ok_or(EngineError::IntegerOverflow)?;
                    sum = sum.checked_add(prod).ok_or(EngineError::IntegerOverflow)?;
                }
                res_num[i][j] = sum;
            }
        }
        Self { num: res_num, den: common_den }.simplify()
    }

    pub fn is_idempotent(&self) -> Result<bool, EngineError> {
        let simplified = self.simplify()?;
        let squared = simplified.checked_mul(&simplified)?;
        Ok(squared == simplified)
    }
}
