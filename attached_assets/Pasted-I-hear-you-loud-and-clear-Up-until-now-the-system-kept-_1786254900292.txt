```rust
use quick_xml::de::from_str;
use quick_xml::events::Event;
use quick_xml::reader::Reader;
use serde::{Deserialize, Serialize};

/// Zero-allocation streaming extractor for EndToEndId using a reusable buffer.
pub fn extract_end_to_end_id<'a>(xml: &'a [u8]) -> Option<&'a str> {
    let mut reader = Reader::from_reader(xml);
    reader.config_mut().trim_text(true);
    let mut buf = Vec::new();
    let mut in_end_to_end = false;

    loop {
        match reader.read_event_into(&mut buf) {
            Ok(Event::Start(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = true;
            }
            Ok(Event::Text(e)) if in_end_to_end => {
                return core::str::from_utf8(e.as_ref()).ok();
            }
            Ok(Event::End(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = false;
            }
            Ok(Event::Eof) | Err(_) => break,
            _ => {}
        }
        buf.clear();
    }
    None
}

use quick_xml::events::Event;

/// Zero-allocation streaming extractor for EndToEndId using a reusable buffer
pub fn extract_end_to_end_id<'a>(xml: &'a [u8]) -> Option<&'a str> {
    let mut reader = Reader::from_reader(xml);
    reader.config_mut().trim_text(true);
    let mut buf = Vec::new();
    let mut in_end_to_end = false;

    loop {
        match reader.read_event_into(&mut buf) {
            Ok(Event::Start(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = true;
            }
            Ok(Event::Text(e)) if in_end_to_end => {
                return core::str::from_utf8(e.as_ref()).ok();
            }
            Ok(Event::End(e)) if e.name().as_ref() == b"EndToEndId" => {
                in_end_to_end = false;
            }
            Ok(Event::Eof) | Err(_) => break,
            _ => {}
        }
        buf.clear();
    }
    None
}
