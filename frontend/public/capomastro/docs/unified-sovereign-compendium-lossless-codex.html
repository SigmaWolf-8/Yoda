pub const EXCHANGE_NYSE: u8 = 1;

pub const EXCHANGE_TSX: u8 = 3;

    pub quantity: i64,

pub fn verify_market_feed(feed: &[u8]) -> Option<FfmvRecord> {
    if feed.len() < 56 { return None; }

    let isin_head: [u8; 8] = feed[0..8].try_into().ok()?;
    let quantity = i64::from_be_bytes(feed[8..16].try_into().ok()?);
    let price = i64::from_be_bytes(feed[16..24].try_into().ok()?);
    let timestamp = i64::from_be_bytes(feed[24..32].try_into().ok()?);
    let bid = i64::from_be_bytes(feed[32..40].try_into().ok()?);
    let ask = i64::from_be_bytes(feed[40..48].try_into().ok()?);
    let volume = u32::from_be_bytes(feed[48..52].try_into().ok()?);
    let exchange_code = feed[52];
    let unqualified_flag = feed[53];

    let exchange_valid = match exchange_code {
        EXCHANGE_NYSE | EXCHANGE_NASDAQ | EXCHANGE_TSX | EXCHANGE_LSE => true,
        _ => false,
    };
    if !exchange_valid || volume <= 100_000 || bid <= 0 || ask <= bid {
        return None;
    }

    let spread = (ask - bid).checked_mul(1000)?;
    if spread >= bid || unqualified_flag != FFMV_TRUE {
        return None;
    }

    Some(FfmvRecord {
        isin_head,
        quantity,
        price,
        timestamp,
        bid,
        ask,
        volume,
        exchange_code,
        auditor_hash_head: EXPECTED_AUDITOR_FINGERPRINT,
        unqualified_flag,
    })
}
